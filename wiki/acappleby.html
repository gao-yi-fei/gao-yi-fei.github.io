<script>

// 配置

const FORUM_URL = 'http://scp-wiki-mc.wikidot.com/forum/t-17592917/acappleby';

// 多个备用代理，按优先级尝试

const PROXY_SERVERS = [

'https://api.allorigins.win/raw?url=',

'https://corsproxy.io/?',

'https://api.codetabs.com/v1/proxy?quest='

];

// 缓存帖子数据

let postsCache = [];

let currentProxyIndex = 0;

let isFetching = false;

// 获取元素

const appleBtn = document.getElementById('appleBtn');

const postContainer = document.getElementById('postContainer');

const infoText = document.getElementById('infoText');

// 显示加载状态

function showLoading() {

postContainer.innerHTML = `

<div class="loading-indicator">

<div class="spinner"></div>

<p>正在获取讨论区数据...</p>

<p style="font-size: 0.9rem; margin-top: 10px;">尝试代理服务器 ${currentProxyIndex + 1}/${PROXY_SERVERS.length}</p>

`;

}

// 显示错误信息

function showError(message) {

postContainer.innerHTML = `

<div class="error-message">

<h3>😕 出错了</h3>

<div style="margin-top: 15px;">

</div>

</div>

`;

}

// 显示帖子

function showPost(post) {

<button class="btn btn-secondary" onclick="refreshPosts()">🔄 刷新数据</button>

</div>

</div>

`;

postContainer.innerHTML = postHtml;

}

// 显示备用帖子（当无法获取真实数据时）

function showFallbackPost() {

const fallbackPosts = [

{

author: 'melicssa1354',

time: '2026-02-22 20:28',

content: '恭喜AC！<br>稍等片刻，我需要采集组织样本。'

},

{

// 随机显示一条帖子

function showRandomPost() {

if (postsCache.length === 0) {

fetchForumPosts();

return;

}

const randomIndex = Math.floor(Math.random() * postsCache.length);

showPost(postsCache[randomIndex]);

infoText.innerHTML = `已加载 ${postsCache.length} 条帖子 · 当前显示第 ${randomIndex + 1} 条 · <a href="${FORUM_URL}" target="_blank">查看原讨论区</a>`;

}

// 尝试下一个代理服务器

function retryWithNextProxy() {

currentProxyIndex = (currentProxyIndex + 1) % PROXY_SERVERS.length;

fetchForumPosts();

}

// 刷新帖子数据

function refreshPosts() {

postsCache = []; // 清空缓存，强制重新获取

fetchForumPosts();

}

// 获取论坛帖子

async function fetchForumPosts() {

if (isFetching) return;

isFetching = true;

appleBtn.classList.add('loading');

showLoading();

try {

const proxyUrl = PROXY_SERVERS[currentProxyIndex] + encodeURIComponent(FORUM_URL);

const controller = new AbortController();

const timeoutId = setTimeout(() => controller.abort(), 10000); // 10秒超时

const response = await fetch(proxyUrl, {

signal: controller.signal

});

clearTimeout(timeoutId);

if (!response.ok) {

throw new Error(`HTTP错误: ${response.status}`);

}

const htmlText = await response.text();

const posts = parsePosts(htmlText);

if (posts.length === 0) {

throw new Error('未找到帖子内容，可能是页面结构不匹配');

}

postsCache = posts;

showRandomPost();

} catch (error) {

console.error('获取数据失败:', error);

if (error.name === 'AbortError') {

showError('请求超时，网络连接较慢或代理服务器无响应。');

} else if (currentProxyIndex < PROXY_SERVERS.length - 1) {

// 自动尝试下一个代理

setTimeout(() => {

retryWithNextProxy();

}, 500);

} else {

showError(`无法获取数据: ${error.message}<br>已尝试所有代理服务器。`);

}

} finally {

isFetching = false;

appleBtn.classList.remove('loading');

}

}

// 解析帖子内容

function parsePosts(html) {

const posts = [];

const parser = new DOMParser();

const doc = parser.parseFromString(html, 'text/html');

// 多种选择器尝试，适应不同的Wikidot模板

const selectors = [

'div.forum-post',

'div.post',

'div.thread-post',

'[id^="post-"]',

'.forum-post-container',

'.post-container',

'.message'

];

let postElements = [];

for (const selector of selectors) {

postElements = doc.querySelectorAll(selector);

if (postElements.length > 0) {

console.log(`使用选择器 "${selector}" 找到 ${postElements.length} 个帖子`);

break;

}

}

// 如果没找到，尝试更通用的方法

if (postElements.length === 0) {

// 查找所有包含用户信息的div

const allDivs = doc.querySelectorAll('div');

postElements = Array.from(allDivs).filter(div => {

const text = div.textContent || '';

return text.includes('Posted by') || text.includes('发表于') ||

(div.innerHTML.includes('printuser') && div.innerHTML.includes('odate'));

});

}

postElements.forEach((element, index) => {

try {

// 提取作者信息

let author = '某个苹果';

const authorSelectors = [

'.printuser a',

'.printuser img[alt]',

'.user-link',

'.username',

'a[href*="user:"]',

'a[href*="user-info"]'

];

for (const selector of authorSelectors) {

const authorElem = element.querySelector(selector);

if (authorElem) {

author = authorElem.textContent ||

authorElem.getAttribute('alt') ||

authorElem.getAttribute('title') ||

author;

break;

}

}

// 提取时间信息

let time = '时间未知';

const timeSelectors = [

'.odate',

'time',

'.post-time',

'.published',

'.date',

'.odate.time'

];

for (const selector of timeSelectors) {

const timeElem = element.querySelector(selector);

if (timeElem) {

time = timeElem.textContent ||

timeElem.getAttribute('datetime') ||

timeElem.getAttribute('title') ||

time;

break;

}

}

// 提取内容

let content = '（无内容）';

const contentSelectors = [

'.post-body',

'.content',

'.message',

'.forum-post-body',

'.post-content',

'.description'

];

for (const selector of contentSelectors) {

const contentElem = element.querySelector(selector);

if (contentElem) {

// 清理内容中的脚本和样式

const cleanElem = contentElem.cloneNode(true);

const scripts = cleanElem.querySelectorAll('script, style, iframe');

scripts.forEach(el => el.remove());

content = cleanElem.innerHTML;

break;

}

}

// 如果通过选择器没找到内容，尝试获取元素的文本内容

if (content === '（无内容）') {

content = element.textContent.substring(0, 500) + '...';

}

posts.push({

id: index,

author: author.trim(),

time: time.trim(),

content: content.trim()

});

} catch (error) {

console.warn(`解析第 ${index} 个帖子时出错:`, error);

}

});

// 如果没有解析到任何帖子，创建一个提示帖子

if (posts.length === 0) {

posts.push({

id: 0,

author: '解析器',

time: new Date().toLocaleDateString(),

content: '未能自动解析到帖子内容。<br>这可能是因为讨论区页面结构特殊，或需要使用不同的解析方法。'

});

}

return posts;

}

function escapeHtml(text) {

if (!text) return '';

const div = document.createElement('div');

div.textContent = text;

return div.innerHTML;

}

        const PROXY_SERVERS = [

        function showRandomPost() {

            if (postsCache.length === 0) {

                fetchForumPosts();

                return;

            }

            showPost(postsCache[randomIndex]);

        // 尝试下一个代理服务器

        function retryWithNextProxy() {

            currentProxyIndex = (currentProxyIndex + 1) % PROXY_SERVERS.length;

            fetchForumPosts();

        }



        // 刷新帖子数据

        function refreshPosts() {

            postsCache = []; // 清空缓存，强制重新获取

            fetchForumPosts();

        // 获取论坛帖子

        async function fetchForumPosts() {

            if (isFetching) return;

            

            isFetching = true;

            appleBtn.classList.add('loading');

            showLoading();

            

            try {

                const proxyUrl = PROXY_SERVERS[currentProxyIndex] + encodeURIComponent(FORUM_URL);

                const controller = new AbortController();

                const timeoutId = setTimeout(() => controller.abort(), 10000); // 10秒超时

                

                const response = await fetch(proxyUrl, { 

                    signal: controller.signal 

                });

                

                clearTimeout(timeoutId);

                

                if (!response.ok) {

                    throw new Error(`HTTP错误: ${response.status}`);

                }

                

                const htmlText = await response.text();

                const posts = parsePosts(htmlText);

                

                if (posts.length === 0) {

                    throw new Error('未找到帖子内容，可能是页面结构不匹配');

                }

                

                postsCache = posts;

                showRandomPost();

                

            } catch (error) {

                console.error('获取数据失败:', error);

                

                if (error.name === 'AbortError') {

                    showError('请求超时，网络连接较慢或代理服务器无响应。');

                } else if (currentProxyIndex < PROXY_SERVERS.length - 1) {

                    // 自动尝试下一个代理

                    setTimeout(() => {

                        retryWithNextProxy();

                    }, 500);

                } else {

                    showError(`无法获取数据: ${error.message}<br>已尝试所有代理服务器。`);

                }

            } finally {

                isFetching = false;

                appleBtn.classList.remove('loading');

            }

        }



        // 解析帖子内容

        function parsePosts(html) {

            const posts = [];

            const parser = new DOMParser();

            const doc = parser.parseFromString(html, 'text/html');

            

            // 多种选择器尝试，适应不同的Wikidot模板

            const selectors = [

                'div.forum-post',

                'div.post',

                'div.thread-post',

                '[id^="post-"]',

                '.forum-post-container',

                '.post-container',

                '.message'

            ];

            

            let postElements = [];

            

            for (const selector of selectors) {

                postElements = doc.querySelectorAll(selector);

                if (postElements.length > 0) {

                    console.log(`使用选择器 "${selector}" 找到 ${postElements.length} 个帖子`);

                    break;

                }

            }

            

            // 如果没找到，尝试更通用的方法

            if (postElements.length === 0) {

                // 查找所有包含用户信息的div

                const allDivs = doc.querySelectorAll('div');

                postElements = Array.from(allDivs).filter(div => {

                    const text = div.textContent || '';

                    return text.includes('Posted by') || text.includes('发表于') || 

                           (div.innerHTML.includes('printuser') && div.innerHTML.includes('odate'));

                });

            }

            

            postElements.forEach((element, index) => {

                try {

                    // 提取作者信息

                    let author = '某个苹果';

                    const authorSelectors = [

                        '.printuser a',

                        '.printuser img[alt]',

                        '.user-link',

                        '.username',

                        'a[href*="user:"]',

                        'a[href*="user-info"]'

                    ];

                    

                    for (const selector of authorSelectors) {

                        const authorElem = element.querySelector(selector);

                        if (authorElem) {

                            author = authorElem.textContent || 

                                    authorElem.getAttribute('alt') || 

                                    authorElem.getAttribute('title') || 

                                    author;

                            break;

                        }

                    }

                    

                    // 提取时间信息

                    let time = '时间未知';

                    const timeSelectors = [

                        '.odate',

                        'time',

                        '.post-time',

                        '.published',

                        '.date',

                        '.odate.time'

                    ];

                    

                    for (const selector of timeSelectors) {

                        const timeElem = element.querySelector(selector);

                        if (timeElem) {

                            time = timeElem.textContent || 

                                   timeElem.getAttribute('datetime') || 

                                   timeElem.getAttribute('title') || 

                                   time;

                            break;

                        }

                    }

                    

                    // 提取内容

                    let content = '（无内容）';

                    const contentSelectors = [

                        '.post-body',

                        '.content',

                        '.message',

                        '.forum-post-body',

                        '.post-content',

                        '.description'

                    ];

                    

                    for (const selector of contentSelectors) {

                        const contentElem = element.querySelector(selector);

                        if (contentElem) {

                            // 清理内容中的脚本和样式

                            const cleanElem = contentElem.cloneNode(true);

                            const scripts = cleanElem.querySelectorAll('script, style, iframe');

                            scripts.forEach(el => el.remove());

                            content = cleanElem.innerHTML;

                            break;

                        }

                    }

                    

                    // 如果通过选择器没找到内容，尝试获取元素的文本内容

                    if (content === '（无内容）') {

                        content = element.textContent.substring(0, 500) + '...';

                    }

                    

                    posts.push({

                        id: index,

                        author: author.trim(),

                        time: time.trim(),

                        content: content.trim()

                    });

                    

                } catch (error) {

                    console.warn(`解析第 ${index} 个帖子时出错:`, error);

                }

            });

            

            // 如果没有解析到任何帖子，创建一个提示帖子

            if (posts.length === 0) {

                posts.push({

                    id: 0,

                    author: '解析器',

                    time: new Date().toLocaleDateString(),

                    content: '未能自动解析到帖子内容。<br>这可能是因为讨论区页面结构特殊，或需要使用不同的解析方法。'

                });

            }

            

            return posts;

        }
